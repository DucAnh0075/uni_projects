25inline Vec3 combineLights(const Vec3& E1, const Vec3& E2) {
26 return Vec3::Ones() - (Vec3::Ones() - E1).cwiseProduct(Vec3::Ones() - E2);

30inline Vec3 computeReflection(const Vec3& incident, const Vec3& normal) {

35inline Vec3 computeRefraction(const Vec3& incident, const Vec3& normal,
36 float n1, float n2) {
37 float ratio = n1 / n2;
38 float cosI = -normal.dot(incident);
39 float sinT2 = ratio * ratio * (1.0f - cosI * cosI);
40 if (sinT2 > 1.0f) return Vec3::Zero(); // Total internal reflection
41 float cosT = std::sqrt(1.0f - sinT2);
42 return ratio * incident + (ratio * cosI - cosT) * normal;
43}

46inline Vec3 getSurfaceNormal(const std::shared_ptr<Object>& obj,
47 const Vec3& hitPoint) {
48 if (auto* sphere = dynamic_cast<const Sphere*>(obj.get())) {
49 Vec3 center(sphere->getPos().x, sphere->getPos().y, sphere->getPos().z);
50 return (hitPoint - center).normalized();
51 } else if (auto* plane = dynamic_cast<const HalfSpace*>(obj.get())) {
52 return plane->getNormal();
53 } else if (auto* cube = dynamic_cast<const Cube*>(obj.get())) {
54 Vec3 center(cube->getPos().x, cube->getPos().y, cube->getPos().z);
55 Vec3 local = hitPoint - center;
56 float ax = std::abs(local.x()), ay = std::abs(local.y()),
57 az = std::abs(local.z());
58 float maxAxis = std::max({ax, ay, az});
59 if (ax == maxAxis) return Vec3((local.x() > 0) ? 1 : -1, 0, 0);
60 if (ay == maxAxis) return Vec3(0, (local.y() > 0) ? 1 : -1, 0);
61 return Vec3(0, 0, (local.z() > 0) ? 1 : -1);
62 }
63 return Vec3(0, 1, 0); // Default
64}

68 static Vec3 traceRay(const Scene& scene, const Vec3& rayOrigin,
69 const Vec3& rayDirection, int depth,
70 RayTracer::PixelInfoPool& pixelPool,
71 RayTracer::MaterialCache& materialCache,
72 RayTracer::IntersectionCache& intersectionCache,
73 int maxDepth = RayTracer::DEFAULT_RAY_DEPTH) {
74 if (depth >= maxDepth) {
75 return scene.backgroundColor;
76 }
77
78 float t;
79 PixelInfo& hitInfo = pixelPool.acquire();
80 std::shared_ptr<Object> hitObject;
81
82 if (!scene.trace(rayOrigin, rayDirection, hitInfo, t, hitObject) ||
83 !hitObject) {
84 pixelPool.release(hitInfo);
85 return scene.backgroundColor;
86 }
87
88 Vec3 hitPoint(hitInfo.pos.x, hitInfo.pos.y, hitInfo.pos.z);
89 Vec3 normal = getSurfaceNormal(hitObject, hitPoint);
90 Vec3 viewDir = (rayOrigin - hitPoint).normalized();
91 const Material& mat = hitObject->getMaterial();
92
93 // Ensure normal points toward viewer
94 if (normal.dot(viewDir) < 0) {
95 normal = -normal;
96 }
97
98 // Base color from object (normalized to [0,1])
99 Vec3 objectColor = RayTracer::colorToVec3(hitInfo.color);
100 Vec3 ambient = objectColor * RayTracer::AMBIENT_INTENSITY;
101
102 // Direct lighting (diffuse + specular)
103 Vec3 diffuse = Vec3::Zero();
104 Vec3 specular = Vec3::Zero();
105
106 for (const auto& light : scene.lights) {
107 Vec3 lightVec = light.getLightVector(hitPoint);
108 float distToLight = (light.getType() == Light::Type::Point)
109 ? lightVec.norm()
110 : std::numeric_limits<float>::infinity();
111
112 if (distToLight <= RayTracer::MIN_INTERSECTION_DISTANCE) continue;
113
114 Vec3 lightDir = lightVec.normalized();
115 float ndotl = std::max(0.0f, normal.dot(lightDir));
116 if (ndotl <= 0.0f) continue;
117
118 // Shadow testing
119 Vec3 shadowOrig = hitPoint + RayTracer::SHADOW_BIAS * normal;
120 bool blocked = scene.isOccluded(
121 shadowOrig, lightDir,
122 std::isfinite(distToLight) ? distToLight
123 : std::numeric_limits<float>::infinity(),
124 RayTracer::SHADOW_BIAS);
125 if (blocked) continue;
126
127 float lightFactor = light.getAttenuatedIntensity(hitPoint);
128
129 Vec3 diffuseContrib = objectColor * ndotl * lightFactor;
130 diffuse = combineLights(diffuse, diffuseContrib);
131 Vec3 reflectDir = computeReflection(-lightDir, normal);
132 float RdotV = std::max(0.0f, reflectDir.dot(viewDir));
133 float Ks = 0.25f;
134 float alpha = 32.0f;
135 Vec3 specularContrib =
136 Vec3(1.0f, 1.0f, 1.0f) * Ks * std::pow(RdotV, alpha) * lightFactor;
137 specular = combineLights(specular, specularContrib);
138 }
139
140
141 Vec3 reflection = Vec3::Zero();
142 Vec3 refraction = Vec3::Zero();
143
144 if (depth < maxDepth) {
145 // Calculate reflection for reflective materials
146 if (mat.Kr.maxCoeff() > 0.01f) {
147 Vec3 reflectDir = computeReflection(-viewDir, normal);
148 Vec3 reflectedColor =
149 traceRay(scene, hitPoint + normal * 0.001f, reflectDir, depth + 1,
150 pixelPool, materialCache, intersectionCache, maxDepth);
151 reflection = Vec3(mat.Kr.x() * reflectedColor.x(),
152 mat.Kr.y() * reflectedColor.y(),
153 mat.Kr.z() * reflectedColor.z());
154 }
155
156 // Calculate refraction for transparent materials
157 if (mat.Kt.maxCoeff() > 0.01f) {
158 Vec3 refractDir =
159 computeRefraction(-viewDir, normal, 1.0f, mat.refractive_index);
160 if (refractDir != Vec3::Zero()) {
161 Vec3 refractedColor =
162 traceRay(scene, hitPoint - normal * 0.001f, refractDir, depth + 1,
163 pixelPool, materialCache, intersectionCache, maxDepth);
164 refraction = Vec3(mat.Kt.x() * refractedColor.x(),
165 mat.Kt.y() * refractedColor.y(),
166 mat.Kt.z() * refractedColor.z());
167 }
168 }
169 }
170
171 // Simple combination like original (no logarithmic for now)
172 Vec3 finalColor = ambient + diffuse + specular + reflection + refraction;
173
174
175 pixelPool.release(hitInfo);
176
177 // Clamp to reasonable range
178 return finalColor.cwiseMin(1.0f).cwiseMax(0.0f);
179 }

249 auto projectToScreen = [&](const Vec3& P, int& outX, int& outY) -> bool {
250 Vec3 O = RayTracer::vec4ToVec3(observer);
251 float denom = (P.z() - O.z());
252 if (std::abs(denom) < 1e-6f) return false;
253 float t_proj = (screen.origin.z() - O.z()) / denom;
254 if (t_proj <= 0.0f) return false;
255 Vec3 S = O + t_proj * (P - O);
256 int sx = (int)((S.x() - screen.origin.x()) / screen.pixel_size_x);
257 int sy = (int)((S.y() - screen.origin.y()) / screen.pixel_size_y);
258 if (sx < 0 || sx >= width || sy < 0 || sy >= height) return false;
259 outX = sx;
260 outY = height - 1 - sy;
261 return true;
262 };

268 Vec3 rayDirection = rayGrid.getRayDirection(x, y);

275 RayTracer::clampVector(finalColor, 0.0f, 1.0f));

279 cv::Vec3b((uchar)finalColor.z(), (uchar)finalColor.y(),

283 y % RayTracer::RAY_VISUALIZATION_STRIDE == 0) {
284 cv::Point startPt((int)(camera.x() * screen.dpi_x + width / 2),

286 cv::Point endPt(x, height - 1 - y);
287 cv::line(image, startPt, endPt, cv::Scalar(0, 0, 255), 1);

310#endif

315 cv::circle(image, cv::Point(lx, fly), 18, cv::Scalar(0, 0, 0), 2);

319 cv::Scalar(0, 0, 0), 2);
320 }
321
322 if (showPlaneFlow) {
323 Vec3 n = planeNormal;
324 int offset = planeFlowStride / 2;
325 size_t drawn = 0;
326 for (int y = offset; y < height; y += planeFlowStride) {
327 for (int x = offset; x < width; x += planeFlowStride) {
328 float px = screen.origin.x() + (x + 0.5f) * screen.pixel_size_x;
329 float py = screen.origin.y() + (y + 0.5f) * screen.pixel_size_y;
330 float pz = screen.origin.z();
331 Vec4 pixel_center(px, py, pz, 1);
332 Vec4 ray_dir = pixel_center - observer;
333 ray_dir.w() = 0;
334 Vec3 ray_dir3 = RayTracer::vec4ToVec3(ray_dir);
335 ray_dir3.normalize();
336 float t;
337 PixelInfo info;
338 if (!scene.trace(camera, ray_dir3, info, t)) continue;
339 if (std::abs(
340 planeNormal.dot(Vec3(info.pos.x, info.pos.y, info.pos.z)) -
341 planeOffset) > RayTracer::LARGE_EPSILON)
342 continue;
343 Vec3 P = camera + ray_dir3 * t;
344 Vec3 L = hasPointLight
345 ? (pointLightPos - P)
346 : (hasDirLight ? (-dirLightDir) : Vec3(0, 0, 0));
347 Vec3 n_used = (n.dot(L) < 0.0f) ? -n : n;
348 Vec3 L_tan = L - n_used * n_used.dot(L);
349 float len = L_tan.norm();
350 if (len < RayTracer::MIN_INTERSECTION_DISTANCE) continue;
351 Vec3 dir = L_tan / len;
352 float stepWorld = 0.8f;
353 Vec3 Q = P + stepWorld * dir;

357 cv::Point(x1, y1), cv::Scalar(255, 0, 255), 1,
358 cv::LINE_AA, 0, 0.25);
359 if (++drawn >= RayTracer::MAX_FLOW_ARROWS) break;
360 }
361 if (drawn >= RayTracer::MAX_FLOW_ARROWS) break;
362 }
363 }
364
365 if (showPositionMarkers) {
366 for (const auto& obj : scene.objects) {
367 Position pos = obj->getPos();
368 Vec3 posVec(pos.x, pos.y, pos.z);
369 int px, py;
370 if (projectToScreen(posVec, px, py)) {
371 cv::circle(image, cv::Point(px, py), 6, cv::Scalar(255, 255, 0), 2);
372 cv::circle(image, cv::Point(px, py), 3, cv::Scalar(0, 255, 255), -1);
373
374 // Show position coordinates as text
375 std::string posText = "(" + std::to_string((int)pos.x) + "," +
376 std::to_string((int)pos.y) + "," +
377 std::to_string((int)pos.z) + ")";

379 cv::FONT_HERSHEY_SIMPLEX, 0.4, cv::Scalar(255, 255, 0),

381 }
382 }
383 }
