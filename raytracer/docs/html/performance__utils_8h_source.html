25 void initialize(const Screen& screen, const Eigen::Vector4f& observer,
26 int width, int height) {
27 width_ = width;
28 height_ = height;
29
30 rayDirections.reserve(width * height);
31 rayOrigins.reserve(width * height);
32
33 Eigen::Vector3f observer3(observer.x(), observer.y(), observer.z());
34
35 for (int y = 0; y < height; ++y) {
36 for (int x = 0; x < width; ++x) {
37 // Calculate pixel center in world coordinates
38 float px = screen.origin.x() + (x + 0.5f) * screen.pixel_size_x;
39 float py = screen.origin.y() + (y + 0.5f) * screen.pixel_size_y;
40 float pz = screen.origin.z();
41 Eigen::Vector4f pixel_center(px, py, pz, 1);
42
43 // Calculate ray direction
44 Eigen::Vector4f ray_dir = pixel_center - observer;
45 ray_dir.w() = 0;
46 Eigen::Vector3f ray_dir3(ray_dir.x(), ray_dir.y(), ray_dir.z());
47 ray_dir3.normalize();
48
49 rayDirections.push_back(ray_dir3);
50 rayOrigins.push_back(observer3);
51 }
52 }
53 }
54
56 const Eigen::Vector3f& getRayDirection(int x, int y) const {
57 return rayDirections[y * width_ + x];
58 }

61 const Eigen::Vector3f& getRayOrigin(int x, int y) const {
