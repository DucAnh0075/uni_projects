102inline std::optional<std::pair<float, float>> solveQuadratic(float a, float b,
103 float c) {
104 if (std::abs(a) < EPSILON) {
105 // Linear equation: bx + c = 0
106 if (std::abs(b) < EPSILON) {
107 return std::nullopt; // No solution or infinite solutions
108 }
109 float x = -c / b;
110 return std::make_pair(x, x);
111 }
112
113 float discriminant = b * b - 4 * a * c;
114 if (discriminant < 0) {
115 return std::nullopt; // No real solutions
116 }
117
118 float sqrtDisc = std::sqrt(discriminant);
119 float x1 = (-b - sqrtDisc) / (2 * a);
120 float x2 = (-b + sqrtDisc) / (2 * a);
121
122 return std::make_pair(std::min(x1, x2), std::max(x1, x2));
123}

126inline std::optional<std::pair<float, float>> rayAABBIntersection(
127 const Eigen::Vector3f& rayOrigin, const Eigen::Vector3f& rayDir,
128 const Eigen::Vector3f& boxMin, const Eigen::Vector3f& boxMax) {
129 float tNear = -std::numeric_limits<float>::infinity();
130 float tFar = std::numeric_limits<float>::infinity();
131
132 for (int i = 0; i < 3; ++i) {
133 if (std::abs(rayDir[i]) < EPSILON) {
134 // Ray is parallel to slab
135 if (rayOrigin[i] < boxMin[i] || rayOrigin[i] > boxMax[i]) {
136 return std::nullopt; // Ray misses slab entirely
137 }
138 } else {
139 float t1 = (boxMin[i] - rayOrigin[i]) / rayDir[i];
140 float t2 = (boxMax[i] - rayOrigin[i]) / rayDir[i];
141
142 if (t1 > t2) std::swap(t1, t2);
143
144 tNear = std::max(tNear, t1);
145 tFar = std::min(tFar, t2);
146
147 if (tNear > tFar) {
148 return std::nullopt; // No intersection
149 }
150 }
151 }
152
153 return std::make_pair(tNear, tFar);
154}
